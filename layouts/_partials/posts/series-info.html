{{- /* Returns one entry per series this page belongs to (with 2+ posts): dict "term" (series term page) "pages" (series posts, oldest first) "index" (this page's 0-based position in "pages"). "pages" $pages "index" $idx) -}} {{- end -}} {{- end -}} {{- return $out -}}
